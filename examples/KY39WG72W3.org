* Magit worktrees
*** magit-worktree
Suggest =~/src/wt/<repo>/<branch>= as the default location when creating a new
worktree via magit's native commands (=magit-worktree-checkout=,
=magit-worktree-branch=).  Matches the layout used by =my/dispatch-task=.
#+begin_src emacs-lisp
  (defun my/magit-read-worktree-directory (prompt branch)
    "Read a directory for a new worktree under ~/src/wt/<repo>/<branch>.
  Suitable for `magit-read-worktree-directory-function'.  BRANCH is nil
  when no branch is being checked out; in that case no initial input is
  offered.  Slashes in branch names are replaced with dashes to match
  magit's built-in readers."
    (let* ((repo (file-name-nondirectory
                  (directory-file-name default-directory)))
           (base (file-name-as-directory
                  (expand-file-name repo my/worktree-root))))
      (read-directory-name
       prompt base nil nil
       (and branch (string-replace "/" "-" branch)))))

  (setq magit-read-worktree-directory-function
        #'my/magit-read-worktree-directory)
#+end_src

*** forge-worktree
Forge uses its own customization point with a different signature: the function
takes the pullreq as its only argument and must return a directory path.  We
mirror =my/magit-read-worktree-directory= so =forge-checkout-worktree= drops
new worktrees under =~/src/wt/<repo>/<branch>= too.
#+begin_src emacs-lisp
  (with-eval-after-load 'forge
    (defun my/forge-checkout-worktree-read-directory (pullreq)
      "Suggest ~/src/wt/<repo>/<branch> for `forge-checkout-worktree'.
  Signature matches `forge-checkout-worktree-read-directory-function':
  takes PULLREQ and returns a directory path."
      (pcase-let* (((eieio number head-ref) pullreq)
                   (repo (file-name-nondirectory
                          (directory-file-name default-directory)))
                   (base (file-name-as-directory
                          (expand-file-name repo my/worktree-root)))
                   (branch (forge--pullreq-branch-internal pullreq))
                   (initial (if (string-match-p "\\`pr-[0-9]+\\'" branch)
                                (number-to-string number)
                              (string-replace "/" "-" head-ref)))
                   (path (read-directory-name
                          (format "Checkout #%s in new worktree: " number)
                          base nil nil initial)))
        (when (equal path "")
          (user-error "The empty string isn't a valid path"))
        path))

    (setq forge-checkout-worktree-read-directory-function
          #'my/forge-checkout-worktree-read-directory))
#+end_src

*** perspective integration
When magit or forge creates a worktree, they finish by visiting the new
directory, which projectile recognizes as a new project -- but nothing creates
a matching perspective.  Mirror =projectile-persp-bridge= so the new worktree
lands in its own perspective, the way =projectile-persp-switch-project= would.
=forge-checkout-worktree= calls =magit-worktree-checkout= internally, so
advising the magit commands covers the forge flow too.
#+begin_src emacs-lisp
  (defun my/worktree-persp-switch (&rest _)
    "Create or switch to a perspective matching the current projectile project."
    (when (and (bound-and-true-p persp-mode)
               (projectile-project-p))
      (persp-switch (projectile-project-name))))

  (with-eval-after-load 'magit
    (advice-add 'magit-worktree-checkout :after #'my/worktree-persp-switch)
    (advice-add 'magit-worktree-branch   :after #'my/worktree-persp-switch))
#+end_src

      "Smart agent-shell dispatcher.

     With prefix ARG, delegate to `agent-shell' with the prefix.
     If in an agent-shell buffer, switch to the previous buffer without closing the window.
     If the project's agent shell is visible but not focused, focus it.

         (arg

         (shell-buffers

    (interactive)

*** agent-shell dashboard
Tile all agent-shell buffers in a grid. Useful for monitoring multiple
concurrent agents.
#+begin_src emacs-lisp
  (defun my/tile-agent-shells ()
    "Tile all agent-shell buffers in a grid in the current frame."
    (interactive)
    (let ((bufs (seq-filter
                 (lambda (b)
                   (eq (buffer-local-value 'major-mode b) 'agent-shell-mode))
                 (buffer-list))))
      (when (null bufs)
        (user-error "No agent-shell buffers found"))
      (delete-other-windows)
      (let* ((n (length bufs))
             (cols (ceiling (sqrt n)))
             (rows (ceiling (/ (float n) cols))))
        (dotimes (_ (1- rows))
          (split-window-below))
        (balance-windows)
        (let ((row-wins (window-list nil 'no-mini)))
          (dolist (w row-wins)
            (select-window w)
            (dotimes (_ (1- cols))
              (split-window-right))))
        (balance-windows)
        (let ((all-wins (window-list nil 'no-mini)))
          (cl-loop for buf in bufs
                   for win in all-wins
                   do (set-window-buffer win buf))
          (cl-loop for win in (nthcdr n all-wins)
                   do (delete-window win))))))

  (defun my/agent-shell-perspective ()
    "Switch to (or create) an 'agents' perspective with tiled agent-shell buffers."
    (interactive)
    (persp-switch "agents")
    (my/tile-agent-shells))

  (bind-key "C-c M-<SPC>" #'my/agent-shell-perspective)
#+end_src

*** dispatch-task
Pick an existing worktree or create a new one, then start an
agent-shell there. Worktrees live under =~/src/wt/<repo>/=.
#+begin_src emacs-lisp
  (defvar my/worktree-root "~/src/wt/"
    "Root directory for centralized worktrees.")

  (defun my/worktree-repo-name ()
    "Return the basename of the current repo's toplevel directory."
    (when-let ((root (magit-toplevel)))
      (file-name-nondirectory (directory-file-name root))))

  (defun my/worktree-list-external ()
    "Return alist of (PATH . BRANCH) for worktrees under `my/worktree-root' for the current repo."
    (when-let ((repo (my/worktree-repo-name)))
      (let ((prefix (expand-file-name (file-name-concat my/worktree-root repo))))
        (seq-filter
         (lambda (entry)
           (string-prefix-p prefix (car entry)))
         (mapcar (lambda (wt)
                   (let* ((path (car wt))
                          (branch (nth 2 wt)))
                     (cons path (or branch "(detached)"))))
                 (magit-list-worktrees))))))

  (defun my/dispatch-task ()
    "Pick or create a worktree, then start an agent-shell in it."
    (interactive)
    (unless (magit-toplevel)
      (user-error "Not in a git repository"))
    (let* ((repo (my/worktree-repo-name))
           (existing (my/worktree-list-external))
           (new-label "[new worktree]")
           (choices (cons
                     (cons new-label nil)
                     (mapcar (lambda (entry)
                               (let ((dir (file-name-nondirectory
                                           (directory-file-name (car entry))))
                                     (branch (cdr entry)))
                                 (cons (format "%s (%s)" dir branch)
                                       (car entry))))
                             existing)))
           (choice (completing-read "Worktree: " choices nil t))
           (worktree-path
            (or (alist-get choice choices nil nil #'string=)
                (let* ((branch (read-string "Branch name: "))
                       (path (expand-file-name
                              (file-name-concat my/worktree-root repo branch))))
                  (make-directory (file-name-directory path) t)
                  (magit-worktree-branch path branch "HEAD")
                  (unless (file-exists-p path)
                    (user-error "Failed to create worktree at %s" path))
                  path))))
      (persp-switch (file-name-nondirectory (directory-file-name worktree-path)))
      (agent-shell--new-shell :location worktree-path)))
#+end_src
